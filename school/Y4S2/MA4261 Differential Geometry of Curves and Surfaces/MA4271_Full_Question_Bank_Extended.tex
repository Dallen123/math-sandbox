\section{Curves and Frenet--Serret Theory}
%===========================================================

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=6pt]

\item Let $\alpha(t)=(a\cos t, a\sin t, bt)$ with $a,b>0$. Reparametrise by arc length and compute the Frenet trihedron $\{\mbf{t},\mbf{n},\mbf{b}\}$. Show $\kp$ and $\tp$ are constant and express them in terms of $a,b$.

\item \hard Prove both directions: a unit-speed curve $\alpha(s)$ is a plane curve if and only if $\tp(s)\equiv 0$.

\item For a non-unit-speed curve $\alpha(t)$, derive the formulas
\[
  \mbf{t}=\frac{\alpha'}{|\alpha'|},\quad \mbf{b}=\frac{\alpha'\times\alpha''}{|\alpha'\times\alpha''|},\quad \kp=\frac{|\alpha'\times\alpha''|}{|\alpha'|^3}, \quad \tp = \frac{(\alpha'\times\alpha'')\cdot\alpha'''}{|\alpha'\times\alpha''|^2}.
\]

\item Compute $\kp(t)$ and $\tp(t)$ for $\alpha(t)=(t,t^2,t^3)$. Verify $\kp(0)=2$ and find $\lim_{t\to 0}\tp(t)$.

\item \hard \textbf{Lancret's Theorem.} Prove a curve is a generalised helix (constant angle with a fixed direction) if and only if $\kp/\tp$ is constant.

\item Let $\alpha(s)$ be unit-speed. Derive the third-order Taylor expansion
\[
  \alpha(s)=\alpha(0)+s\,\mbf{t}(0)+\frac{s^2}{2}\kp(0)\mbf{n}(0)+\frac{s^3}{6}\bigl(\kp'(0)\mbf{n}(0)+\kp(0)\tp(0)\mbf{b}(0)-\kp(0)^2\mbf{t}(0)\bigr)+O(s^4).
\]

\item Let $\alpha(s)$ be unit-speed and lie on a sphere of radius $R$. Prove
\[
  R^2=\frac{1}{\kp^2}+\left(\frac{1}{\tp}\cdot\frac{d}{ds}\frac{1}{\kp}\right)^2.
\]

\item \hard \textbf{Darboux Vector.} Show the Frenet equations $\mbf{t}'=\kp\mbf{n}$, $\mbf{n}'=-\kp\mbf{t}+\tp\mbf{b}$, $\mbf{b}'=-\tp\mbf{n}$ can be written uniformly as $\mbf{v}'=\mbf{d}\times\mbf{v}$ for $\mbf{v}\in\{\mbf{t},\mbf{n},\mbf{b}\}$, where $\mbf{d}=\tp\mbf{t}+\kp\mbf{b}$.

\item \hard Derive the Frenet--Serret equations from scratch by differentiating the orthonormality relations $\mbf{t}\cdot\mbf{t}=1$, $\mbf{t}\cdot\mbf{n}=0$, $\mbf{b}\cdot\mbf{b}=1$, etc.

\item Write the Frenet--Serret system as the matrix ODE
\[
  \begin{pmatrix}\mbf{t}'\\\mbf{n}'\\\mbf{b}'\end{pmatrix}=\begin{pmatrix}0&\kp&0\\-\kp&0&\tp\\0&-\tp&0\end{pmatrix}\begin{pmatrix}\mbf{t}\\\mbf{n}\\\mbf{b}\end{pmatrix}.
\]
Explain why the matrix is skew-symmetric, and show this follows from $\{\mbf{t},\mbf{n},\mbf{b}\}$ being orthonormal.

\item \hard Prove: if $\mathbf{b}(s)=\mathbf{c}$ is constant, then $\alpha(s)$ lies in a plane perpendicular to $\mathbf{c}$. Conversely, if $\alpha$ lies in a plane with unit normal $\mathbf{c}$, then $\mathbf{b}=\pm\mathbf{c}$ and $\tp\equiv 0$.

\item Let $\alpha(t)=(e^t\cos t, e^t\sin t, e^t)$. Compute the arc length from $t=0$ to $T$, find $\mbf{t}(t)$, and show $\kp(t)$ is constant.

\item \hard \textbf{Fundamental Theorem of Curve Theory.} State the theorem. Outline the existence proof via the Frenet ODE system and prove uniqueness using the energy function $f(s)=|\mbf{t}_1-\mbf{t}_2|^2+|\mbf{n}_1-\mbf{n}_2|^2+|\mbf{b}_1-\mbf{b}_2|^2$.

\item \hard Suppose $\alpha(s)$ and $\beta(s)$ have the same $\kp(s)>0$ and $\tp(s)$ and identical Frenet frames at $s=0$. Define $f(s)=|\mbf{t}_1-\mbf{t}_2|^2+|\mbf{n}_1-\mbf{n}_2|^2+|\mbf{b}_1-\mbf{b}_2|^2$. Show $f'\equiv 0$, hence $f\equiv 0$, and integrate to conclude $\alpha=\beta$.

\item A rigid motion is $R(\mathbf{x})=A\mathbf{x}+\mathbf{v}$ with $A\in SO(3)$. Let $\beta(s)=R(\alpha(s))$. Show $\kp_\beta=\kp_\alpha$ and $\tp_\beta=\tp_\alpha$.

\item Compute $\kp(t)$ and $\tp(t)$ for the curve $\alpha(t)=(e^t,-\cosh t, t)$ (Tutorial 11 exam curve).

\item Let $\alpha(t)=(3t-t^3,3t^2,3t+t^3)$. Using non-unit-speed formulas, compute $\kp$ and $\tp$ and verify $\kp=-\tp=1/(3(1+t^2)^2)$.

\item For a curve $\alpha(t)$ not passing through the origin, suppose $\alpha(t_0)$ is the point of the trace closest to the origin and $\alpha'(t_0)\neq\mathbf{0}$. Prove $\alpha(t_0)\perp\alpha'(t_0)$.

\item \hard Let $\alpha:I\to\R^3$ and $\mathbf{v}\in\R^3$ be fixed. Assume $\alpha'(t)\perp\mathbf{v}$ for all $t\in I$ and $\alpha(0)\perp\mathbf{v}$. Prove $\alpha(t)\perp\mathbf{v}$ for all $t\in I$.

\item Suppose all normals $\mbf{n}(s)$ of a unit-speed curve pass through a fixed point. Prove the trace is contained in a circle.

\item A regular parametrised curve $\alpha$ has all its tangent lines passing through a fixed point. Prove the trace is a (segment of a) straight line. Does this hold if regularity fails?

\item Let $\alpha:I\to\R^3$ be unit-speed with $\kp(s)\neq 0$. Let $P$ be a plane containing the tangent line at $s_0$ such that $\alpha(J)$ has points on both sides of $P$ for any neighbourhood $J$ of $s_0$. Prove $P$ is the osculating plane at $s_0$.

\item \hard \textbf{Bertrand Curves.} Show $\alpha$ and $\beta$ are Bertrand mates if and only if $a\kp+b\tp=1$ for constants $a,b$.

\item Compute $\kp$ and $\tp$ for the curve $\alpha(\theta)=(r(\theta)\cos\theta, r(\theta)\sin\theta, 0)$ in polar coordinates. Show that the arc-length formula gives $\int_a^b\sqrt{r^2+(r')^2}\,d\theta$.

\item Derive $\mbf{n}'(s)=-\kp(s)\mbf{t}(s)-\tp(s)\mbf{b}(s)$ by differentiating $\mbf{n}=\mbf{b}\wedge\mbf{t}$ using the cross-product rule $(\mbf{u}\wedge\mbf{v})'=\mbf{u}'\wedge\mbf{v}+\mbf{u}\wedge\mbf{v}'$.

\item \hard For a unit-speed curve $\alpha(s)$, show $\alpha(s)\cdot\mbf{b}(0)=O(s^3)$, meaning the curve departs from its osculating plane only to third order. (Expand $\alpha(s)\cdot\mbf{b}(0)$ in Taylor series and use $\mbf{n}(0)\perp\mbf{b}(0)$.)

\item Let $f(s)=\alpha(s)\cdot\mbf{c}$ for a fixed unit vector $\mbf{c}$ and unit-speed $\alpha$. Compute $f'(s)=\mbf{t}\cdot\mbf{c}$ and $f''(s)=\kp\,\mbf{n}\cdot\mbf{c}$. Deduce: if $\kp>0$ and $\tp\equiv 0$, then $\alpha$ lies in the plane $\{\mathbf{x}:\mathbf{x}\cdot\mbf{b}(0)=\alpha(0)\cdot\mbf{b}(0)\}$.

\item A circular helix has constant $\kp_0>0$ and constant $\tp_0$. Starting from the Frenet ODEs, reduce to a second-order ODE for $\mbf{t}(s)$, solve to show $\mbf{t}(s)$ traces a circle on $S^2$, and integrate to recover the helix.

\item For a unit-speed ellipse $\alpha(s)=(a\cos(s/L), b\sin(s/L), 0)$ (suitably reparametrised), show $\kp'(t)=0$ has exactly four solutions in $[0,2\pi)$, verifying the Four-Vertex Theorem.

\item \hard \textbf{Spherical Image.} For a unit-speed curve $\alpha(s)$ with $\kp>0$, the spherical image is $\gamma(s)=\mbf{t}(s)\in S^2$. Show $\kp_\gamma(s)=\sqrt{\kp^2+(\kp'/\kp)^2}$ and give a geometric interpretation.

\item Given vectors $v\neq 0$ and $w$, show there exists $u$ with $u\wedge v=w$ if and only if $v\perp w$. Is $u$ unique? If not, describe the general solution.

\item Compute the arc-length parametrisation of $\alpha(t)=(t-\sin t, 1-\cos t, 4\sin(t/2))$.

\item \hard Let $S$ be a rigid motion sending $\alpha$ to $\beta$, i.e.\ $S(\mathbf{x})=A\mathbf{x}+\mathbf{v}$ with $A\in SO(3)$. Show that $S$ is an isometry of $\R^3$ and deduce that two curves with the same $\kp$ and $\tp$ differ by a rigid motion.

\item Compute the curvature and torsion of the tubular surface curve $\alpha(s,v)=\gamma(s)+r(\mbf{n}(s)\cos v+\mbf{b}(s)\sin v)$ along a fixed $v$, expressing them in terms of $\kp_\gamma$, $\tp_\gamma$, $r$, and $v$.

\item Show: if $\kp\equiv 0$ then $\alpha$ is a straight line regardless of $\tp$. Separately, if $\kp>0$ and $\tp\equiv 0$, show $\alpha$ lies in a plane.

\item \hard Find the principal curvatures $k_1,k_2$ and the Gaussian curvature $K$ of the tubular surface $\mathbf{x}(s,v)=\alpha(s)+r(\mbf{n}(s)\cos v+\mbf{b}(s)\sin v)$ around a unit-speed curve $\alpha$ with curvature $\kp$ and torsion $\tp$.

\item Prove the product rule for cross products: $(\mathbf{u}(t)\wedge\mathbf{v}(t))'=\mathbf{u}'(t)\wedge\mathbf{v}(t)+\mathbf{u}(t)\wedge\mathbf{v}'(t)$.

\item \hard Show that a rigid motion $R(x)=Ax+v$ with $A\in O(3)$, $\det A=-1$ (a reflection) reverses the sign of $\tp$ while preserving $\kp$. What happens to the Frenet frame?

\item \hard Prove that two smooth curves with $k_1(s)=k_2(s)>0$ and $\tau_1(s)=\tau_2(s)$ for all $s\in\R$ that share the same Frenet frame at $s=0$ are identical. Use this to give a careful proof of uniqueness in the Fundamental Theorem.

\item \hard \textbf{Isoperimetric Inequality.} State $L^2\geq 4\pi A$ for a simple closed plane curve. Sketch the Fourier series proof and characterise equality.

\item \hard \textbf{Signed Curvature and Umlaufsatz.} For a simple closed plane curve $\alpha(t)=(x(t),y(t))$, define $\kp_s=(x'y''-y'x'')/|\alpha'|^3$. State Hopf's Umlaufsatz and verify the total signed curvature is $\pm 2\pi$ for (a) a circle, (b) an ellipse.

\item \hard \textbf{Fenchel's Theorem.} State Fenchel's Theorem and use it to deduce that a closed curve with $\kp<2\pi/L$ cannot be convex.

\item \hard For a closed space curve $\alpha:[0,L]\to\R^3$, the total torsion is $\int_0^L\tp(s)\,ds$. Show this equals the total rotation of the binormal vector around the curve. Compute it for the standard unknot (a circle) and argue geometrically that it is non-zero for the $(2,1)$-torus knot.

\end{enumerate}

\section{Regular Surfaces and Parametrisations}
%===========================================================

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=6pt]

\item State the Pre-image Theorem. Use it to show the ellipsoid $x^2/a^2+y^2/b^2+z^2/c^2=1$ is a regular surface.

\item Show the cone $z^2=x^2+y^2$ is not a regular surface by examining $\nabla f$ at the origin. Explain the geometric failure.

\item For the surface of revolution $\mathbf{x}(u,v)=(f(v)\cos u, f(v)\sin u, v)$ with $f(v)>0$, state and verify the three conditions for a coordinate function. Why is a second coordinate function needed?

\item \hard Show stereographic projection of $S^2\setminus\{N\}$ onto $\R^2$ is a diffeomorphism by exhibiting the explicit formulas and verifying smoothness in both directions.

\item Let $S=\{f(x,y,z)=c\}$ with $\nabla f\neq\mathbf{0}$ on $S$. Show $\nabla f(p)\perp T_pS$ for all $p\in S$.

\item Compute $E,F,G$ for the catenoid $\mathbf{x}(u,v)=(c\cosh v\cos u, c\cosh v\sin u, cv)$.

\item Compute the surface area of the torus $\mathbf{x}(u,v)=((R+r\cos u)\cos v, (R+r\cos u)\sin v, r\sin u)$ and obtain $A=4\pi^2 Rr$.

\item Show the graph $\{(x,y,f(x,y))\}$ of any smooth $f:\R^2\to\R$ is a regular surface. Exhibit the coordinate function and verify all three conditions.

\item Show $\mathbf{x}(u,v)=(u+v, u-v, uv)$ defines a regular surface and compute $E,F,G$.

\item Compute $E,F,G$ for the helicoid $\mathbf{x}(u,v)=(v\cos u, v\sin u, cu)$ and find the unit normal $\mathbf{N}$.

\item \hard Show the transition map $\mathbf{y}^{-1}\circ\mathbf{x}$ between two overlapping coordinate functions is a diffeomorphism, by showing it is smooth and using the Inverse Function Theorem for the local inverse.

\item Define $T_pS$ via velocity vectors of smooth curves through $p$. Show $T_pS$ is a 2-dimensional subspace of $\R^3$ and equals the column space of $d\mathbf{x}_p$.

\item \hard Using the Implicit Function Theorem, show that $f(x,y,z)=0$ with $\partial f/\partial z\neq 0$ can locally be written as $z=g(x,y)$.

\item Show the area formula $\iint|\mathbf{x}_u\times\mathbf{x}_v|\,du\,dv$ is independent of the choice of orientation-preserving parametrisation.

\item Prove the identity $\|\mathbf{x}_u\wedge\mathbf{x}_v\|^2=EG-F^2$ and use it to compute the area element $dA=\sqrt{EG-F^2}\,du\,dv$.

\item Parametrise the M\"obius strip. Show it is not orientable by demonstrating the unit normal cannot be defined continuously on the whole strip.

\item Define $d\varphi_p:T_pS_1\to T_{\varphi(p)}S_2$ and show it is a well-defined linear map independent of the choice of curve used to represent $\mathbf{v}_1$.

\item \hard Prove: if $\varphi:S_1\to S_2$ is a bijective smooth map with $d\varphi_p$ bijective for all $p$, and if $\varphi^{-1}$ is also smooth (i.e.\ $\varphi$ is a diffeomorphism), then $\varphi$ pulls back coordinate functions to coordinate functions.

\item \hard Show that $\mathbf{x}(u,v)=(\sin 2u,\sin u, v)$ on $(-\pi,\pi)\times(-1,1)$ is a smooth injection with $\rank(d\mathbf{x})=2$, yet $\mathbf{x}^{-1}$ is not continuous. Explain why this does not violate the regularity theorem (identify the missing hypothesis).

\item For the surface of revolution of $x=z^2+1$ about the $z$-axis, verify it is a regular surface using the Pre-image theorem. Consider $S$ and the function $\mathbf{x}(u,v)=((v^2+1)\cos u,(v^2+1)\sin u,v)$: compute $\mathbf{x}_u$ and $\mathbf{x}_v$ at a general point and verify $d\mathbf{x}$ has rank 2.

\item \hard Let $f:\R^3\to\R$ be smooth. Show $f^{-1}(a)$ is a regular surface whenever $\nabla f\neq\mathbf{0}$ on $f^{-1}(a)$. Apply this to show $S=\{x^9+y^3+(z-2)^2+xy^4+y(z-2)^6=4\}$ is a regular surface (Tutorial 11 exam question).

\item Let $S$ be a regular surface, $f:S\to\R$ smooth at $p$ using chart $\mathbf{x}$, and let $\mathbf{y}$ be another chart covering $p$. Show $(f\circ\mathbf{y})=(f\circ\mathbf{x})\circ h$ where $h=\mathbf{x}^{-1}\circ\mathbf{y}$ is smooth, and conclude smoothness of $f$ is chart-independent.

\item Let $\varphi:S\to\bar{S}$ be a diffeomorphism. Show that if $\mathbf{x}:U\to S$ is a coordinate function of $S$, then $\mathbf{y}:=\varphi\circ\mathbf{x}:U\to\bar{S}$ is a coordinate function of $\bar{S}$.

\item \hard Let $F:V\subset\R^3\to\R^3$ be a smooth map and $S\subset V$ a regular surface with $F(S)\subset\tilde{S}$ for another regular surface $\tilde{S}$. Show the restriction $\varphi=F|_S:S\to\tilde{S}$ is smooth. Apply to the antipodal map $\varphi(p)=-p$ on $S^2$.

\item For the paraboloid $z=x^2+y^2$ with $\mathbf{x}(u,v)=(u,v,u^2+v^2)$: compute $E,F,G$; find the unit normal $\mathbf{N}$; and set up and evaluate the area integral over $\{x^2+y^2\leq R^2\}$.

\item Let $S_1,S_2$ be two regular surfaces intersecting along a regular curve $C=\alpha(s)$ (arc-length). Let $k_1,k_2$ be the normal curvatures of $C$ in $S_1,S_2$ respectively, and $\theta$ the angle between the normal vectors $N_1,N_2$. Prove $k^2\sin^2\theta=k_1^2+k_2^2-2k_1k_2\cos\theta$ where $k$ is the curvature of $C$ (Tutorial 8 / HWK 6).

\item Define the angle between two tangent vectors $\mathbf{v}_1=a\mathbf{x}_u+b\mathbf{x}_v$ and $\mathbf{v}_2=c\mathbf{x}_u+d\mathbf{x}_v$. Derive the formula for this angle in terms of $E,F,G,a,b,c,d$.

\item \hard Compute the first fundamental form for $P(u,v)=(u\cos v, u\sin v, \varphi(u)+av)$ (a generalised helicoid). Show the $u$-curves and $v$-curves are orthogonal if and only if $\varphi'=0$.

\item For the cone $\mathbf{x}(u,v)=(u\sin\alpha\cos v, u\sin\alpha\sin v, u\cos\alpha)$ with $u>0$, $0<v<2\pi$: compute $E,F,G$ and show a curve $\mathbf{x}(c\exp(v\sin\alpha\coth\beta),v)$ intersects each generator $v=\text{const}$ at constant angle $\beta$.

\item Show the area of a regular tube of radius $r$ around a unit-speed curve $\alpha$ of length $L$ is $2\pi r L$ (HWK 5).

\item If the first fundamental form is $I=du^2+(u^2+a^2)dv^2$, find the arc lengths of the edges and interior angles of the triangle bounded by $C_1:u=\tfrac{a}{2}v^2$, $C_2:u=-\tfrac{a}{2}v^2$, $C_3:v=1$.

\item Let $\phi:S_1\to S_2$ be a bijective smooth map. Give an example where $\phi$ is not a diffeomorphism because $\phi^{-1}:S_2\to S_1$ is not smooth. Explain (Tutorial 7 Qn 3).

\item \hard Let $\phi:S_1\to S_2$ be a bijective smooth map such that $d\phi:T_pS_1\to T_{\phi(p)}S_2$ is a bijection for all $p\in S_1$. Show $\phi^{-1}$ is smooth, i.e.\ $\phi$ is a diffeomorphism (Tutorial 7 Qn 4).

\item For a smooth vector field $\mathbf{w}$ on $S$, and a smooth function $h:S\to\R$, define $\mathbf{w}_p(h)=\frac{d}{dt}h(\alpha(t))|_{t=0}$ where $\alpha'(0)=\mathbf{w}(p)$. Show this is independent of the choice of curve $\alpha$, and prove the Leibniz rule $\mathbf{w}_p(hg)=g(p)\mathbf{w}_p(h)+h(p)\mathbf{w}_p(g)$ (Tutorial 11 Qn 8).

\item \hard Let $\mathbf{w}$ be a smooth vector field on $S$. Show that $\phi:S\to\R$ defined by $\phi(p)=\mathbf{w}_p(h)$ is a smooth function on $S$ for each smooth $h:S\to\R$ (Tutorial 11 Qn 8(iii)).

\item State the definition of a smooth 2-dimensional manifold (Hausdorff, second-countable, with smooth atlas). Show every regular surface in $\R^3$ is a smooth 2-manifold. Give an example of a 2-manifold not embeddable in $\R^3$.

\item Compute $E,F,G$ for $S^2$ in spherical coordinates $\mathbf{x}(\theta,\varphi)=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$. Show $F=0$ and deduce the arc length of the latitude circle $\theta=\theta_0$ is $2\pi\sin\theta_0$.

\item \hard Let $F:\R^3\to\R^3$ be smooth and let $S$ be a regular surface with $F(S)\subset\tilde S$. The lecture uses this to prove smoothness of the restriction. Carefully explain where the Inverse Function Theorem is applied in this argument.

\item Let $\mathbf{w}:S\to\R^3$ be a smooth vector field on a regular surface. Define what it means for $\mathbf{w}$ to be a smooth vector field and show the components $(a(u,v),b(u,v))$ in $\mathbf{w}(\mathbf{x}(u,v))=a\mathbf{x}_u+b\mathbf{x}_v$ transform by the Jacobian of the transition map when you change coordinates (Tutorial 10 Qn 2).

\item \hard For two coordinate functions $\mathbf{x}:U\to S$ and $\mathbf{y}:U'\to S$ with $\mathbf{x}(U)=\mathbf{y}(U')$, let $\Phi=\mathbf{y}^{-1}\circ\mathbf{x}:U\to U'$. If the component functions $a(u,v),b(u,v)$ of a vector field are smooth, show that the component functions $c(\xi,\eta),d(\xi,\eta)$ in the $\mathbf{y}$-chart, which satisfy $(c,d)^T=d\Phi\cdot(a,b)^T$, are also smooth (Tutorial 10 Qn 2(ii), 2005/06 exam).

\item \hard For a compact regular surface $S\subset\R^3$, show $S$ has at least one point of positive Gaussian curvature by considering the point farthest from the origin.

\item Let $S_1$ be the $xy$-plane. Let $S_-=\{(x,y)\in S_1:x<0\}$ and $S_+=\{(x,y)\in S_1:x>0\}$. Give an example of a bijection $\phi:S_1\to S_2$ (not a diffeomorphism) such that $\phi|_{S_-}$ and $\phi|_{S_+}$ are both diffeomorphisms. Also give an example of a diffeomorphism $\psi:S_1\to S_3$ such that $\psi|_{S_-}$ is an isometry but $\psi|_{S_+}$ is not (Tutorial 9 Qn 4).

\end{enumerate}

\section{Second Fundamental Form and Curvatures}
%===========================================================

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=6pt]

\item \hard Show the Weingarten map $dN_p:T_pS\to T_pS$ is self-adjoint: prove $\langle dN(\mathbf{x}_u),\mathbf{x}_v\rangle=\langle\mathbf{x}_u,dN(\mathbf{x}_v)\rangle$ by differentiating $\langle N,\mathbf{x}_u\rangle=0$ with respect to $v$ and vice versa.

\item Derive the formula $K=(eg-f^2)/(EG-F^2)$ from the matrix of $dN$ in the basis $\{\mathbf{x}_u,\mathbf{x}_v\}$.

\item Compute $K$ and $H$ for the sphere $x^2+y^2+z^2=R^2$ using both the Weingarten map and the formula. Verify $K=1/R^2$, $H=1/R$.

\item Find the second fundamental form coefficients $e,f,g$ for the helicoid $\mathbf{x}(u,v)=(v\cos u,v\sin u,cu)$. Show $H=0$ (the helicoid is minimal).

\item \hard Prove: if every point of a connected surface $S$ is umbilic ($\kp_1=\kp_2$ everywhere), then $S$ is contained in a plane or a sphere.

\item Compute $K$ for the Enneper surface $\mathbf{x}(u,v)=(u-u^3/3+uv^2, v-v^3/3+vu^2, u^2-v^2)$ and show $K\leq 0$ everywhere with $K=0$ only at the origin.

\item \hard \textbf{Euler's Theorem.} Let $\mathbf{e}_1,\mathbf{e}_2$ be orthonormal principal directions with principal curvatures $\kp_1,\kp_2$. Show $\kp_n(\theta)=\kp_1\cos^2\theta+\kp_2\sin^2\theta$ where $\theta$ is the angle the tangent direction makes with $\mathbf{e}_1$.

\item \hard From Euler's Theorem, derive that $H=\frac{1}{2\pi}\int_0^{2\pi}\kp_n(\phi)\,d\phi$, i.e.\ mean curvature is the average of normal curvatures over all directions (Tutorial 11 Qn 6(iii)).

\item Let $S$ be the graph $z=f(x,y)$. At a critical point $f_x=f_y=0$, show $K=f_{xx}f_{yy}-f_{xy}^2=\det(\text{Hess}_f)$. Apply to $z=x^2-y^2$ (hyperbolic) and $z=x^2+y^2$ (elliptic).

\item \hard \textbf{Meusnier's Theorem.} Prove all curves through $p$ with the same tangent direction have the same normal curvature $\kp_n=\mathrm{II}(\mathbf{v},\mathbf{v})$, regardless of how they curve in the normal direction.

\item Compute the second fundamental form coefficients $e,f,g$ for the torus $\mathbf{x}(u,v)=((R+r\cos u)\cos v,(R+r\cos u)\sin v,r\sin u)$ and find $K(u,v)=\cos u/(r(R+r\cos u))$.

\item \hard Prove a unit-speed curve is a line of curvature if and only if $N'(s)=-\kp_i\alpha'(s)$ (Rodrigues' Formula).

\item Define asymptotic curves. Show they exist only at non-elliptic points. Determine the asymptotic directions on $z=xy$ and verify they are the coordinate lines (Tutorial 9 / HWK 9).

\item Find the lines of curvature and asymptotic curves of the helicoid $x=v\cos u$, $y=v\sin u$, $z=cu$ (HWK 9 Qn 2).

\item Determine the asymptotic curves and lines of curvature of $z=xy$. Verify your answers agree with the general theory (HWK 9 Qn 3).

\item \hard Show at a hyperbolic point, the principal directions bisect the asymptotic directions (HWK 6 Qn 1).

\item Show the sum of the normal curvatures in any two orthogonal directions at a point equals $\kp_1+\kp_2=2H$ (HWK 6 Qn 2).

\item Compute the Gauss map $N(u,v)=(-f_u,-f_v,1)/\sqrt{1+f_u^2+f_v^2}$ for the graph $z=f(x,y)$. At a critical point, simplify $N$ and compute $e,f,g$.

\item \hard Derive the matrix expression for $[dN]_{\mathcal{B},\mathcal{B}}$ where $\mathcal{B}=\{\mathbf{x}_u,\mathbf{x}_v\}$:
\[
  [dN]_{\mathcal{B},\mathcal{B}} = -\begin{pmatrix}E&F\\F&G\end{pmatrix}^{-1}\begin{pmatrix}e&f\\f&g\end{pmatrix}.
\]
Hence obtain $K=\det(dN)$ and $H=-\tfrac{1}{2}\mathrm{tr}(dN)$ in terms of $e,f,g,E,F,G$.

\item For the surface $\mathbf{x}(u,v)=(u,v,u^2+v^2)$ (paraboloid), compute $dN(\mathbf{x}_u)$ and $dN(\mathbf{x}_v)$, express them as linear combinations of $\mathbf{x}_u,\mathbf{x}_v$, and find $[dN]_{\mathcal{B},\mathcal{B}}$ (Tutorial 7 Qn 1 continuation).

\item At the point $\mathbf{p}=\mathbf{x}(u,v)$ on the paraboloid $z=1+x^2+y^2$, find the principal curvatures for general $(u,v)$. Describe where $k_1=k_2$ (Tutorial 7 Qns 1b--1c).

\item \hard Let $S$ be a regular surface. If $F=f=0$ at a point, show the principal curvatures are $k_1=e/E$ and $k_2=g/G$ (Tutorial 8 Qn 1).

\item \hard \textbf{Gauss Map and Spherical Image.} Show the image $N\circ\alpha$ by the Gauss map of a regular curve $\alpha:I\to S$ containing no planar or parabolic points is a regular curve on $S^2$. If $C=\alpha(I)$ is a line of curvature with curvature $k$ and normal curvature $k_n$, prove $k=|k_n k_N|$ where $k_N$ is the curvature of the spherical image (HWK 6 Qn 3).

\item \hard Prove: the Gauss map area element satisfies $N^*dA_{S^2}=K\,dA_S$. For the sphere $S^2_R$, verify the total image area is $4\pi$. For a cylinder $x^2+y^2=r^2$, explain why $N$ maps to a great circle and reconcile with $K=0$.

\item Compute $K$ and $H$ for the catenoid $\mathbf{x}(u,v)=(c\cosh v\cos u, c\cosh v\sin u, cv)$. Verify $H=0$ directly and show $\kp_1=-\kp_2$.

\item For a surface of revolution $\mathbf{x}(u,v)=(f(v)\cos u, f(v)\sin u, g(v))$ with $(f')^2+(g')^2=1$: derive $K$ and $H$ in terms of $f,g$ and their derivatives. Verify for the sphere ($f=\cos v$, $g=\sin v$) that $K=1$.

\item \hard \textbf{Codazzi--Mainardi Equations.} Derive the equations $e_v-f_u=e\Gamma_{12}^1+f(\Gamma_{12}^2-\Gamma_{11}^1)-g\Gamma_{11}^2$ and its companion from $N_{uv}=N_{vu}$. Verify for the sphere.

\item \hard State Bonnet's Theorem and explain why both the Gauss and Codazzi--Mainardi equations are necessary compatibility conditions for the existence of a surface with given first and second fundamental forms.

\item Determine whether a surface $\mathbf{x}(u,v)$ with $E=1$, $F=0$, $G=\cos^2 u$ and $e=\cos^2 u$, $f=0$, $g=1$ exists, by checking the Codazzi--Mainardi equations (HWK 10 Qn 4, adapted).

\item \hard For a curve $\alpha(t)$ on a regular surface $S$, prove that at any point,
\[
  k_n^2 - \tau_g^2 - 2Hk_n + K = 0,
\]
where $k_n$, $\tau_g$, $H$, $K$ are the normal curvature, geodesic torsion, mean curvature, and Gaussian curvature respectively (HWK 12 Qn 2).

\item Find the principal curvatures $k_1,k_2$ and $K$ for the tubular surface $\mathbf{x}(s,v)=\alpha(s)+r(\mbf{n}(s)\cos v+\mbf{b}(s)\sin v)$ around a unit-speed curve $\alpha$ with curvature $\kp$ and torsion $\tp$ (HWK 9 Qn 1). Describe where $K=0$.

\item \hard Let $\phi:S_1\to S_2$ be a local isometry (for the next section, but uses shape operator ideas). Show $\phi$ need not preserve the second fundamental form by constructing an explicit example (e.g.\ a cylinder unrolled to a plane).

\item Consider two spheres $S_1$ (radius 3) and $S_2$ (radius 1) both centred at the origin. Is there a local isometry $\phi:S_1\to S_2$? Justify using $K$ (Tutorial 11 Qn 7).

\end{enumerate}

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=6pt]

\item Define a local isometry $\varphi:S_1\to S_2$. Prove: $\varphi$ is a local isometry if and only if $E_1=E_2\circ\varphi$, $F_1=F_2\circ\varphi$, $G_1=G_2\circ\varphi$ in compatible parametrisations.

\item \hard \textbf{Theorema Egregium.} State the theorem. Explain why $K$ can be expressed entirely in terms of $E,F,G$ and their first and second partial derivatives. Apply the Brioschi formula to verify $K=1$ for the unit sphere in spherical coordinates ($E=1,F=0,G=\sin^2\theta$).

\item Apply the Brioschi formula to the cylinder $x^2+y^2=r^2$ (parametrised as $\mathbf{x}(u,v)=(\cos u,\sin u,v)$; $E=G=1$, $F=0$) and confirm $K=0$.

\item Show the cylinder $x^2+y^2=r^2$ is locally isometric to a plane by exhibiting explicit coordinate functions with the same first fundamental form.

\item \hard Show the catenoid and helicoid are locally isometric via the parametrisations $\mathbf{x}(u,v)=(v\cos u,v\sin u,cu)$ and $\mathbf{y}(u,v)=(c\cosh v\cos u,c\cosh v\sin u,cv)$. Verify their first fundamental forms are identical and compute $K$ for both (HWK 10 Qn 2, Tutorial 9 context).

\item \hard Show the surfaces $P(u,v)=(u\cos v,u\sin v,u+v)$ and $Q(\rho,\theta)=(\rho\cos\theta,\rho\sin\theta,\sqrt{\rho^2-1})$ are locally isometric via $\theta=\tan^{-1}u+v$, $\rho=\sqrt{u^2+1}$ (HWK 10 Qn 2).

\item If the first fundamental form is $I=(a^2+x^2y^2)dx^2+2bxy\,dx\,dy+b^2\,dy^2$ with $a,b\neq 0$, find the Gaussian curvature (HWK 10 Qn 3).

\item \hard Does there exist a surface with $E=1$, $F=0$, $G=\cos^2 u$ and $e=\cos^2 u$, $f=0$, $g=1$? Check the Codazzi--Mainardi compatibility conditions and compute $K$ two ways (from $e,f,g$ and from $E,F,G$ via Brioschi). Are they consistent? (HWK 10 Qn 4.)

\item Prove $K\equiv 0$ implies $S$ is locally isometric to a plane (developable). List three classes of developable surfaces and verify each has $K=0$.

\item \hard A cone (excluding apex) is locally isometric to a flat sector of a plane. Explain what happens to the apex under ``development'' and why the cone is not globally isometric to any open subset of $\R^2$.

\item Give an explicit equiareal map from $S^2$ to a cylinder that is not an isometry (Lambert equal-area projection). Identify which invariant (area vs.\ first fundamental form) distinguishes equiareal from isometric maps.

\item \hard Derive the general Christoffel symbol formula $\Gamma_{ij}^k=\tfrac{1}{2}g^{kl}(\partial_ig_{jl}+\partial_jg_{il}-\partial_lg_{ij})$. Compute all nonzero $\Gamma_{ij}^k$ for $S^2$ in spherical coordinates and verify $\Gamma_{22}^1=-\sin\theta\cos\theta$, $\Gamma_{12}^2=\cot\theta$.

\item Classify as intrinsic or extrinsic: (a) $K$, (b) $H$, (c) arc length, (d) $\kp_1,\kp_2$, (e) Christoffel symbols $\Gamma_{ij}^k$. Show $K=\kp_1\kp_2$ is intrinsic even though $\kp_1,\kp_2$ individually are not. Exhibit two locally isometric surfaces with different $H$.

\item \hard \textbf{Gauss Equation.} State the Gauss equation $R_{1212}=eg-f^2$. Explain what $R_{1212}$ (computed from Christoffel symbols alone) measures, why this proves $K$ is intrinsic, and verify for $S^2$.

\item Show every isometry preserves area. Then construct an explicit equiareal map that is not an isometry.

\item Define the covariant derivative $DV/dt$ along a curve $\gamma$ as the tangential component of $V'$. Show: (i) $\frac{d}{dt}\langle V,W\rangle=\langle DV/dt,W\rangle+\langle V,DW/dt\rangle$ (metric compatibility); (ii) the Levi-Civita connection is torsion-free: $D_{\mathbf{x}_u}\mathbf{x}_v-D_{\mathbf{x}_v}\mathbf{x}_u=\mathbf{0}$.

\item \hard Derive the Koszul formula $2\langle\nabla_XY,Z\rangle=X\langle Y,Z\rangle+Y\langle X,Z\rangle-Z\langle X,Y\rangle+\langle[X,Y],Z\rangle-\langle[Y,Z],X\rangle+\langle[X,Z],Y\rangle$ and use it to show the Levi-Civita connection is uniquely determined by metric compatibility and torsion-free conditions.

\item For the pseudosphere $\mathbf{x}(u,v)=(\mathrm{sech}\,v\cos u,\mathrm{sech}\,v\sin u,v-\tanh v)$: compute $E,F,G$, verify $F=0$, and apply the Brioschi formula to confirm $K=-1$.

\item \hard Define holonomy: the rotation angle $\Delta\theta$ of a vector parallel-transported around a closed curve. For a geodesic triangle on $S^2$ of area $A$, show $\Delta\theta=A/R^2$. For a small geodesic rectangle of area $\Delta A$ on a surface, show $\Delta\theta\approx K\cdot\Delta A$.

\item \hard \textbf{Isometry Group.} Show the set of isometries $S\to S$ forms a group. Identify all isometries of the flat cylinder $x^2+y^2=1$ explicitly. Show the isometry group of $S^2$ is $O(3)$ (not just $SO(3)$), and explain why reflections are included.

\item Construct an isometry $\phi:C\to C$ of the cylinder $C=\{x^2+y^2=1\}$ such that the set of fixed points $\{p\in C:\phi(p)=p\}$ contains exactly 2 points (HWK 10 Qn 1).

\item \hard By the Theorema Egregium, the Gaussian curvature is preserved by local isometries. Use this to give a rigorous proof that no flat map of the sphere can perfectly preserve distances ($K_{S^2}=1/R^2\neq 0=K_{\text{plane}}$). Describe what properties Mercator (conformal) and Lambert (equiareal) projections do preserve.

\item \hard \textbf{Minding's Theorem.} State the theorem and use it to conclude the pseudosphere (with $K=-1$) is locally isometric to the Poincar\'e half-plane (also $K=-1$). What is the geometric significance?

\item Consider the geodesic on the hyperboloid $x^2+y^2-z^2=1$ (upper part $z>0$) starting at a point $p$ making angle $\theta$ with the parallels, where $\cos\theta=1/r$ and $r$ is the distance from $p$ to the $z$-axis. Using Clairaut's relation, show the geodesic approaches asymptotically the parallel $x^2+y^2=1$, $z=0$ (HWK 12 Qn 1).

\item \hard State the Nash Embedding Theorem. Contrast it with the Whitney Embedding Theorem. Explain what it implies for the existence of abstract Riemannian metrics on surfaces.

\item \hard The flat torus $T^2=\R^2/\Z^2$ has $K=0$ everywhere. Show it is locally isometric to a plane but not globally isometric to any open subset of $\R^2$ (topological obstruction). Show also it is not isometric to the standard torus in $\R^3$ (which has varying $K$): what is the obstruction?

\item \hard A parallel vector field $X(s)$ along a simple closed curve $C$ bounding a region $D$ in a surface $S$ satisfies $\varphi(l)-\varphi(0)=\iint_D K\,dA$ where $\varphi(s)$ is the angle from the $u$-curve to $X(s)$. Prove this (HWK 13 Qn 1).

\item \hard Compute the Euler number by Gauss--Bonnet for: (a) an ellipsoid (answer: $\chi=2$); (b) the surface $S=\{(x,y,z)\in\R^3:x^2+y^{10}+z^6=1\}$ (use the Pre-image Theorem to show $S\cong S^2$ topologically, hence $\chi=2$) (HWK 13 Qn 2).

\end{enumerate}

\section{Geodesics and Parallel Transport}
%===========================================================

\begin{enumerate}[label=\textbf{Q\arabic*.}, leftmargin=*, itemsep=6pt]

\item Derive the geodesic equations $\ddot{u}^k+\Gamma_{ij}^k\dot{u}^i\dot{u}^j=0$ from the Euler--Lagrange equations for the energy functional $E[\gamma]=\frac{1}{2}\int g_{ij}\dot{u}^i\dot{u}^j\,dt$. Show minimising energy gives constant-speed geodesics automatically.

\item Show $\frac{d}{dt}|\gamma'|^2=0$ along any geodesic, so speed is constant. Deduce that arc-length parametrised geodesics satisfy the standard geodesic equations with $|\gamma'|=1$.

\item \hard In spherical coordinates $\mathbf{x}(\theta,\varphi)=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$, write out the geodesic equations using the Christoffel symbols $\Gamma_{22}^1=-\sin\theta\cos\theta$ and $\Gamma_{12}^2=\cot\theta$. Verify the equator $\theta(t)=\pi/2$, $\varphi(t)=t$ satisfies them.

\item Show all great circles on $S^2$ are geodesics by noting they can be obtained from the equator by rigid rotations, and rigid rotations preserve geodesics (isometries preserve geodesics).

\item \hard \textbf{Clairaut's Relation.} For a geodesic $\gamma$ on a surface of revolution with axial distance function $r$, prove $r\sin\psi=\text{const}$ where $\psi$ is the angle $\gamma$ makes with the parallels.

\item Apply Clairaut's relation to the torus ($r=R+r_0\cos u$). Identify which latitude circles $u=u_0$ are geodesics. Describe qualitatively the geodesic starting from the outer equator.

\item Consider a geodesic on the hyperboloid $x^2+y^2-z^2=1$ ($z>0$) at a point $p$ with axial distance $r_p$, making angle $\theta$ with the parallels satisfying $\cos\theta=1/r_p$. Using Clairaut, show the geodesic approaches the parallel $r=1$ asymptotically (HWK 12 Qn 1).

\item Define geodesic curvature $\kp_g=\langle\alpha'',N\wedge\alpha'\rangle$ for a unit-speed curve $\alpha$ on $S$. Show $\gamma$ is a geodesic if and only if $\kp_g\equiv 0$. Prove $\kp^2=\kp_n^2+\kp_g^2$.

\item Compute $\kp_g$ for a latitude circle $\theta=\theta_0$ on the unit sphere and verify $\kp_g=\cot\theta_0$. Confirm the equator ($\theta_0=\pi/2$) is a geodesic.

\item \hard Let $S$ contain the origin with tangent plane $V=$ $xy$-plane and normal $N=(0,0,1)$. Let $\alpha(s)$ be unit-speed on $S$ with $\alpha(0)=(0,0,0)$, $\alpha'(0)=(1,0,0)$, and $\beta(s)=(x(s),y(s),0)$ its projection onto $V$. Show $\alpha''(0)=(0,k_g,k_n)$ and $k^2=k_n^2+k_g^2$ (Tutorial 9 Qn 5).

\item \hard Show parallel transport preserves inner products: if $V(t)$ and $W(t)$ are parallel along $\gamma$, then $\frac{d}{dt}\langle V,W\rangle=0$. Conclude parallel transport is an orthogonal map $T_{\gamma(a)}S\to T_{\gamma(b)}S$.

\item Compute the covariant derivatives $D\mathbf{w}/du$ and $D\mathbf{w}/dv$ for the vector field $\mathbf{w}(x,y,z)=(-y,x,0)$ on the sphere, using the stereographic parametrisation $\mathbf{x}(u,v)=R^{-1}(4u,4v,u^2+v^2-4)$ (Tutorial 10 Qn 1).

\item Show that if $V(t)$ and $W(t)$ are vector fields along $\gamma$, then $\frac{d}{dt}\langle V,W\rangle=\langle DV/dt,W\rangle+\langle V,DW/dt\rangle$ (Tutorial 10 Qn 3).

\item \hard A vector field $\mathbf{v}(t)$ along $\alpha(t)$ is parallel if $D\mathbf{v}/dt=\mathbf{0}$, i.e.\ $\mathbf{v}'(t)\perp T_{\alpha(t)}S$. (i) Show the speed $|\mathbf{v}(t)|$ is constant. (ii) If $\mathbf{v}(t):=\alpha'(t)$ is itself parallel along $\alpha$, show $|\alpha'(t)|$ is constant and $\gamma(s):=\alpha(s/c)$ is a geodesic (Tutorial 10 Qn 4).

\item \hard Parallel-transport a vector $V_0\in T_{p_0}S^2$ around the latitude circle $\theta=\theta_0$ from $\varphi=0$ to $\varphi=2\pi$. Show the resulting vector is rotated by angle $2\pi\cos\theta_0$ relative to $V_0$.

\item \hard \textbf{Local Gauss--Bonnet.} State the local formula: for a geodesic polygon $R$ on $S$ with geodesic sides and exterior angles $\theta_i$, $\sum_i\theta_i+\iint_R K\,dA=2\pi$. Apply to show the angle sum of a geodesic triangle on $S^2$ with area $A$ is $\pi+A$ (with $R=1$).

\item Verify Local Gauss--Bonnet for the octant triangle on $S^2$ (three right angles, $\alpha_1=\alpha_2=\alpha_3=\pi/2$, area $A=\pi/2$): check $3\cdot(\pi/2)=\pi+\pi/2$.

\item \hard \textbf{Global Gauss--Bonnet.} State $\iint_S K\,dA=2\pi\chi(S)$. Compute the integral for $S^2$ ($\chi=2$), torus ($\chi=0$), and genus-$g$ surface ($\chi=2-2g$). Deduce no metric on $T^2$ can have $K>0$ everywhere and no metric on $S^2$ can have $K\leq 0$.

\item Show the Christoffel symbols vanish everywhere on the flat torus $\R^2/\Z^2$. Conclude the geodesic equations become $\ddot{u}^k=0$, so all geodesics are images of straight lines. Show a geodesic is closed iff $\dot{u}^1/\dot{u}^2\in\mathbb{Q}$.

\item Show geodesics on the cylinder $x^2+y^2=r^2$ are helices by unrolling the cylinder to a flat strip (an isometry) and pulling back straight lines. Classify all geodesics: vertical lines, horizontal circles, and general helices.

\item \hard Define the exponential map $\exp_p:T_pS\to S$ by $\exp_p(\mathbf{v})=\gamma_{\mathbf{v}}(1)$ where $\gamma_{\mathbf{v}}$ is the geodesic with $\gamma_{\mathbf{v}}(0)=p$, $\gamma_{\mathbf{v}}'(0)=\mathbf{v}$. Show $d(\exp_p)_{\mathbf{0}}=\mathrm{id}$, hence $\exp_p$ is a local diffeomorphism near $\mathbf{0}$. In normal coordinates, verify $g_{ij}(p)=\delta_{ij}$ and $\Gamma_{ij}^k(p)=0$.

\item \hard Define the Jacobi field $J(t)=\partial\gamma_s/\partial s|_{s=0}$ for a one-parameter family of geodesics $\gamma_s$. Derive the Jacobi equation $J''+KJ=0$ (on a surface). For $S^2$ ($K=1$), solve to find $J(t)=A\sin t+B\cos t$. Show Jacobi fields vanish at $t=0$ and $t=\pi$, proving geodesics have conjugate points at distance $\pi$ and are not minimising beyond that.

\item Define geodesic completeness. State the Hopf--Rinow theorem. Prove every compact regular surface $S\subset\R^3$ is geodesically complete by showing a geodesic that cannot be extended would have to escape $S$.

\item \hard Let $C$ be a positively oriented simple closed curve on $S$ bounding a region $D$ contained in a coordinate chart with orthogonal parametrisation $P(u,v)$. Let $X(s)$ be a parallel vector field along $C$ and $\varphi(s)$ the angle from the $u$-curve to $X(s)$. Prove $\varphi(l)-\varphi(0)=\iint_D K\,dA$ (HWK 13 Qn 1).

\item Show that every geodesic is locally minimising (no conjugate points on short segments). Give a concrete example on $S^2$ showing a geodesic of length $>\pi R$ is not the shortest path between its endpoints.

\item \hard Prove the angle sum of a geodesic triangle on a surface of constant curvature $K=k$ with area $A$ equals $\pi+kA$. Apply to conclude: in hyperbolic space ($k=-1$) angle sums are less than $\pi$; in spherical geometry ($k=1$) they exceed $\pi$.

\item For the vector field $\mathbf{w}=(-y,x,0)$ on the unit sphere $S$, verify it is a smooth vector field using the spherical coordinate chart. Compute the covariant derivatives $D\mathbf{w}/du$ and $D\mathbf{w}/dv$ along the coordinate curves (Tutorial 10 Qn 1 full version).

\item \hard \textbf{Geodesic Torsion.} For a unit-speed curve $\alpha(s)$ on $S$ with unit tangent $\mathbf{t}$ and surface normal $N$, the geodesic torsion is $\tau_g=\langle N',\mathbf{t}\wedge N\rangle$. Prove $\tau_g=-\langle dN(\mathbf{t}),\mathbf{t}\wedge N\rangle$. Show a curve is a line of curvature iff $\tau_g\equiv 0$.

\item \hard \textbf{Bonnet's Diameter Theorem.} If $K\geq\delta>0$ everywhere on a complete surface, outline the Jacobi field argument showing $\mathrm{diam}(S)\leq\pi/\sqrt{\delta}$.

\item State and prove: a smooth vector field on $S^2$ must have at least one zero (Hairy Ball Theorem). Use Poincar\'e--Hopf: $\sum_i\mathrm{ind}(p_i)=\chi(S^2)=2$. Show the ``comb'' field (tangent to longitudes) has zeros of index $+1$ at the poles, confirming $1+1=2$.

\item \hard Using the Local Gauss--Bonnet formula applied to a small geodesic circle of radius $\varepsilon$ on $S$, derive the formula $K(p)=\lim_{\varepsilon\to 0}\frac{3}{\pi\varepsilon^4}(2\pi\varepsilon-L(\varepsilon))$ where $L(\varepsilon)$ is the circumference of the geodesic circle. Interpret: positive $K$ means geodesic circles are shorter than Euclidean circles of the same radius.

\end{enumerate}
